    """Summarise respond/azurelab result JSONs (per rep and pooled, nearest rank)."""
    import json, math, sys, collections
    
    def rank(v, p):
        s = sorted(v)
        return s[max(0, min(len(s) - 1, math.ceil(p / 100 * len(s)) - 1))]
    
    def st(v):
        return 'n=%d p50 %.1f p95 %.1f min %.1f max %.1f' % (len(v), rank(v, 50), rank(v, 95), min(v), max(v)) if v else 'n=0'
    
    pool = collections.defaultdict(list)
    buckets = collections.defaultdict(lambda: [0, 0])
    for path in sys.argv[1:]:
        d = json.load(open(path))
        print('##', path.rsplit('/', 1)[-1], d['started'], '->', d['finished'], 'pass', d['pass'], 'load', d['host_load_start'], '/', d['host_load_end'])
        print('rep | handle ms | graph absent ms | first refusal s | last success s | stable refusal s | OK after first refusal / probes | activation first OK s / stable OK s | replay noop | 2nd alert | old token ARM after s')
        for r in d['reps']:
            t, a = r['token_after_alert'], r['activation']
            row = dict(handle=r['handle']['ms'], graph=r['graph']['absent_ms'], first=t['first_reject_ms'], last=t['last_success_ms'],
                       stable=t['stable_reject_ms'] if r['token_after_alert_stable'] else None, act_first=a['first_ok_ms'], act_stable=a['stable_ok_ms'])
            for k, v in row.items():
                if v is not None:
                    pool[k].append(v)
            ok_after = t['successes_after_first_reject']
            pool['ok_frac'].append(ok_after / t['probes'])
            old = r['arm_with_token_issued_before_alert']
            da = r['distinct_alert_same_deleted_credential']
            print('%d | %.1f | %.1f | %.1f | %.1f | %s | %d / %d | %.1f / %.1f | %s | %s %.1f ms | %s after %.0f' % (
                r['rep'], row['handle'], row['graph'], row['first'] / 1e3, (row['last'] or 0) / 1e3,
                '%.1f' % (row['stable'] / 1e3) if row['stable'] is not None else 'not confirmed', ok_after, t['probes'],
                row['act_first'] / 1e3, row['act_stable'] / 1e3, r['replay']['noop'], da['status'], da['ms'], old['result']['status'], old['token_age_s']))
            for p in r['token_probes']:
                if p['ok'] or (p.get('error') == 'invalid_client'):
                    buckets[int(p['at_ms'] // 60000)][0 if p['ok'] else 1] += 1
            for p in r['token_probes'] + r['activation_probes']:
                if not p['ok'] and p.get('error') != 'invalid_client':
                    pool['inconclusive_kinds'].append((p.get('status'), p.get('error')))
    print('## pooled over all reps')
    for k in ('handle', 'graph', 'first', 'last', 'stable', 'act_first', 'act_stable'):
        print(k, '(ms)', st(pool[k]))
    print('ok fraction after first refusal per rep', ['%.2f' % x for x in pool['ok_frac']])
    print('minute after alert: tokens issued / refused (all reps)')
    for m in sorted(buckets):
        print('  %2d-%2d min: %4d / %4d' % (m, m + 1, *buckets[m]))
    print('inconclusive probes', collections.Counter(pool['inconclusive_kinds']))

## script (python3 -I -B orig_criterion.py <json>...)
    """Replay the committed token probes under the criterion before 0e32aa7: a stable refusal is 30 s in which
    every conclusive probe after the first refusal (that follows the last success) was refused, and it must
    come within the 10-minute cap (pollRejection in 0e32aa7^:respond/azurelab/live_test.go). Conclusive means
    a token (ok) or an invalid_client refusal; other failures neither count nor restart, as in the code.
    The committed runs probed every 1 s; the old test probed every 500 ms, so this is the same rule on
    coarser data. For comparison: the current rule (60 s window, 30-min cap) as recorded in the JSON."""
    import json, sys
    WINDOW_MS, CAP_MS = 30_000, 600_000
    def refused(p):
        return p.get("status") in (400, 401) and p.get("error") == "invalid_client"
    for path in sys.argv[1:]:
        d = json.load(open(path))
        print("##", path.rsplit("/", 1)[-1], "poll", d["poll"])
        fails = 0
        for rep in d["reps"]:
            since, verdict = None, None
            for p in rep["token_probes"]:
                done = p["at_ms"] + p["ms"]
                if p["ok"]:
                    since = None
                elif refused(p) and since is None:
                    since = p["at_ms"]
                if since is not None and done - since >= WINDOW_MS:
                    verdict = "meets it: 30 s of refusals from %.1f s, confirmed at %.1f s" % (since / 1e3, done / 1e3)
                    break
                if done > CAP_MS:
                    verdict = "FAILS it: no 30 s of refusals within 600 s (cap reached at %.1f s)" % (done / 1e3)
                    fails += 1
                    break
            later = [p["at_ms"] for p in rep["token_probes"] if p["ok"] and since is not None and p["at_ms"] > since]
            t = rep["token_after_alert"]
            print("rep %d: %s; last success %.1f s; current rule: stable from %.1f s%s" % (
                rep["rep"], verdict, t["last_success_ms"] / 1e3, t["stable_reject_ms"] / 1e3,
                "; a success came after the 30 s window, at %.1f s" % (later[0] / 1e3) if later and verdict.startswith("meets") else ""))
        print("=> %d of %d reps fail the original criterion; the run %s it" % (fails, len(d["reps"]), "FAILS" if fails else "meets"))
