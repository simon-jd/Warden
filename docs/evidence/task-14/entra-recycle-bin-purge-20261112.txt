## driver (python3 -I -B purge.py <ids>)
    """L2: permanently delete Warden's deleted lab apps/SPs from the Entra recycle bin, by exact ID only.
    
    Before each purge: GET /v1.0/directory/deletedItems/<id> and require the expected type and a
    displayName starting with warden-lab-. The deleted-items collection is never listed. Only
    type, id, displayName and deletedDateTime are printed (never credentials or hints). A refused
    purge (permission) is recorded verbatim and stops the step.
    """
    import datetime as dt, json, subprocess, sys
    GRAPH = "https://graph.microsoft.com/v1.0/directory/deletedItems/"
    TYPES = {"application": "#microsoft.graph.application", "servicePrincipal": "#microsoft.graph.servicePrincipal"}
    def now():
        return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"
    def az(*args):
        p = subprocess.run(["az", *args], capture_output=True, text=True, timeout=120)
        return p.returncode, p.stdout, p.stderr.strip()
    def get(i):
        code, out, err = az("rest", "--method", "GET", "--uri", GRAPH + i + "?$select=id,displayName,deletedDateTime")
        if code != 0:
            return None, err
        d = json.loads(out)
        return {"type": d.get("@odata.type"), "id": d.get("id"), "displayName": d.get("displayName"),
                "deletedDateTime": d.get("deletedDateTime")}, None
    done = refused = 0
    for line in open(sys.argv[1]):
        run, kind, i = line.split()
        print(f"### {now()} {run} {kind} {i}")
        item, err = get(i)
        print(f"GET deletedItems/{i} -> " + (json.dumps(item) if item else f"error: {err}"))
        if not item:
            print("skipped: not in the recycle bin (nothing purged)")
            continue
        if item["id"] != i or item["type"] != TYPES[kind] or not str(item["displayName"]).startswith("warden-lab-"):
            print("skipped: type or displayName does not match (nothing purged)")
            continue
        t = now()
        code, out, err = az("rest", "--method", "DELETE", "--uri", GRAPH + i)
        print(f"{t} DELETE deletedItems/{i} -> exit {code}" + (f": {err}" if err else ""))
        if code != 0:
            refused += 1
            if any(s in err for s in ("Authorization_RequestDenied", "Forbidden", "Insufficient privileges", "403")):
                print("STOP: the tenant refused the purge; no further purge attempted")
                break
            continue
        done += 1
        after, err = get(i)
        print(f"{now()} GET deletedItems/{i} after purge -> " + (json.dumps(after) if after else f"error: {err}"))
    print(f"### {now()} purged {done}, refused {refused}")
