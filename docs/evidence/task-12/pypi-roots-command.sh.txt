root.mkdir(mode=0o700)

names=[('pysigma','1.5.1'),('pysigma-backend-opensearch','2.0.3'),('pysigma-backend-elasticsearch','2.1.1')]

            assert response.status==200 and response.geturl()==url
            while True:
                if time.monotonic()>=deadline: raise TimeoutError('absolute metadata deadline')
