"""Tag-derived reporting boundary; initial semantic TDD stub.

Fixtures, conversion, indexed queries and live detection are separate evidence.
"""


def summarize_rules(rules: list[dict]) -> dict:
    """Return a nonempty report structure; actual aggregation is pending."""
    if not rules:
        raise ValueError("coverage input cannot be empty")
    return {
        "rule_count": 0,
        "tagged_rule_count": 0,
        "unassigned_rule_count": 0,
        "technique_count": 0,
        "rules": [],
        "techniques": [],
    }
