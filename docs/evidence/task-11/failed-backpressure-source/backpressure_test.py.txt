"""One finite Vector0.45.0 disk-buffer experiment; synthetic IDs, owned peer only.

Each canonical input JSON+LF is exactly16384B. Baseline320 events/10s, then
32000 target events/100s. Burst first20s peer ACK<=32/s, next60s no ACK,
last20s unrestricted. Pre-stall target excess=(320-32)*20*16384=94371840B;
stall target bytes=314572800B; sum408944640 > disk268435488+96MiB inflight.
Pinned topology source queue=threads2*CHUNK1000; allow pending/pump1000 each,
stdio1024*8KiB plus HTTP/pipe/partial frames, rounded UP to96MiB. This is a
static offered-load bound, not proof actual acceptance reached target rate.
PASS additionally needs >=1s continuous EAGAIN during ACK stall, overlapping
an actual sampled buffer_byte_size>=260000000, and all32320 received IDs.
Short pipe partial writes or occupancy>0 alone cannot pass. No exact-once,
physical-crash, production-source, OpenSearch or real-alert-firing claim.
"""

def phase(now):
    b = state['burst_start']
    if b is None or now < b:
        return 'baseline'
    if now < b+20:
        return 'burst_1x_peer'
    if now < b+80:
        return 'burst_stall'
    if now < b+100:
        return 'burst_release'
    return 'drain'

BASE = {'activity_id': 6, 'category_uid': 4, 'class_uid': 4001,
        'type_uid': 400106, 'severity_id': 0, 'time': 1767225600000,
        'metadata': {'version': '1.3.0', 'product': {'name': 'Task11 Fixture', 'vendor_name': 'Warden'}, 'uid': 'bp-00000'},

        'unmapped': {'padding': ''}}

def line(number):
    BASE['metadata']['uid'] = 'bp-%05d' % number
    b = json.dumps(BASE, sort_keys=True, separators=(',', ':')).encode()+b'\n'
    if len(b) != LINE_BYTES:
        raise ValueError('fixed line size changed')
    return b

class Peer(http.server.HTTPServer):
    timeout = 0.1
    allow_reuse_address = False
    def get_request(self):
        connection, address = super().get_request()
        remaining = END-time.monotonic()
        if remaining <= 0:
            connection.close()
            raise TimeoutError('peer accept deadline')
        connection.settimeout(min(0.2, remaining))
        with lock:
            state['active_connection'] = connection
        return connection, address
    def close_request(self, request):
        with lock:
            if state['active_connection'] is request:
                state['active_connection'] = None
        super().close_request(request)
    def handle_error(self, _request, _address):
        with lock:
            state['error'] = 'owned HTTP peer handler failed'

class Handler(http.server.BaseHTTPRequestHandler):
    protocol_version = 'HTTP/1.1'
    def log_message(self, *_args):
        pass
    def do_POST(self):
        try:
            remaining = END-time.monotonic()
            if remaining <= 0 or stop.is_set():
                raise TimeoutError('peer body deadline')
            self.connection.settimeout(min(0.2, remaining))
            n = int(self.headers.get('Content-Length', '-1'))
            if not 0 < n <= 65536:
                raise ValueError('HTTP body cap/framing')
            body = self.rfile.read(n)
            if len(body) != n:
                raise ValueError('short HTTP body')
            rows = body.splitlines()
            if len(rows) != 1:
                raise ValueError('expected one event/request')
            obj = json.loads(rows[0])
            uid = obj['metadata']['uid']
            if not isinstance(uid, str) or len(uid) != 8 or not uid.startswith('bp-') or not uid[3:].isdigit():
                raise ValueError('unexpected synthetic ID')
            i = int(uid[3:])
            if not 0 <= i < COUNT or obj['unmapped']['padding'] != 'p'*PAD:
                raise ValueError('synthetic payload mismatch')
            if len(json.dumps(obj, sort_keys=True, separators=(',', ':')).encode())+1 != LINE_BYTES:
                raise ValueError('received canonical event size')
            with lock:
                phase_counts[phase(time.monotonic())]['received'] += 1
                if seen[i]:
                    state['duplicates'] += 1
                else:
                    seen[i] = 1
                    state['received'] += 1
                if state['first_received'] is None:
                    state['first_received'] = time.monotonic()-START
            throttled = False
            while not stop.is_set():
                now = time.monotonic()
                if now >= END:
                    raise TimeoutError('peer ACK deadline')
                b = state['burst_start']
                if b is not None and b <= now < b+20:
                    if throttled:
                        break
                    with lock:
                        target = max(state['next_ack'], b)
                        state['next_ack'] = target+1/32
                        state['throttled_acks'] += 1
                    if target > now:
                        stop.wait(min(target-now, max(0.0, END-now)))
                    throttled = True
                    continue
                if b is not None and b+20 <= now < b+80:
                    stop.wait(min(0.05, b+80-now, END-now))
                    continue
                break
            before = time.monotonic()
            if before >= END or stop.is_set():
                raise TimeoutError('peer stopped before ACK')
            b = state['burst_start']
            if b is not None and b+20 <= before < b+80:
                raise RuntimeError('ACK send check inside fixed stall window')
            self.send_response(200 if not stop.is_set() else 503)
            self.send_header('Content-Length', '0')
            self.end_headers()
            now = time.monotonic()
            b = state['burst_start']
            with lock:
                state['ack_checked_count'] += 1
                if b is not None and before < b+80 and now >= b+20:
                    state['ack_window_violation'] = {'before_seconds': before-START, 'after_seconds': now-START}
                    state['error'] = 'ACK send overlaps fixed stall window'
                if b is not None and now < b+20:
                    state['last_ack_before_stall'] = now-START
                    state['last_ack_before_interval'] = [before-START, now-START]
                elif b is not None and now >= b+80 and state['first_ack_after_stall'] is None:
                    state['first_ack_after_stall'] = now-START
                    state['first_ack_after_interval'] = [before-START, now-START]
        except Exception as exc:
            with lock:
                state['error'] = str(exc)[:256]
            self.close_connection = True

def capture(key, data):
    global truncated
    room = 524288-log_sizes[key]
    prefix = data[:room]
    if prefix:
        logs[key].write(prefix)
        log_sizes[key] += len(prefix)
    if len(data) > room:
        truncated = True

def io_once(timeout):
    for key, _mask in sel.select(max(0.0, timeout)):
        if key.data in ('stdout', 'stderr'):
            data = os.read(key.fd, 65536)
            if data:
                capture(key.data, data)
            else:
                sel.unregister(key.fileobj)

def metrics():
    global metric_fd, metric_offset, metric_tail, metric_bytes, metric_samples
    global peak_buffer, last_buffer, last_buffer_events
    path = RUN/'metrics.ndjson'
    if metric_fd is None:
        if not path.exists():
            return
        s = path.lstat()
        if not stat.S_ISREG(s.st_mode) or s.st_uid != 1000 or s.st_nlink != 1:
            raise ValueError('owned metrics type/owner')
        metric_fd = path.open('rb', buffering=0)
    data = metric_fd.read(65536)
    if not data:
        return
    metric_bytes += len(data)
    if metric_bytes > 2097152:
        raise ValueError('metrics2MiB cap')
    metric_tail += data
    rows = metric_tail.split(b'\n')
    metric_tail = rows.pop()
    if len(metric_tail) > 16384:
        raise ValueError('metric line cap')
    for raw in rows:
        m = json.loads(raw)
        if m.get('namespace') != 'vector' or m.get('tags', {}).get('component_id') != 'bp_http':
            raise ValueError('unexpected metrics series')
        name = m['name']
        if name in ('buffer_byte_size', 'buffer_events'):
            value = m['gauge']['value']
            if not isinstance(value, (int, float)) or not math.isfinite(value) or value < 0:
                raise ValueError('invalid occupancy gauge')
            if name == 'buffer_byte_size':
                last_buffer = value
                peak_buffer = max(peak_buffer, value)
                occupancy.append((time.monotonic(), value))
                metric_samples += 1
            else:
                last_buffer_events = value
        elif name == 'buffer_discarded_events_total':
            value = m['counter']['value']
            if not isinstance(value, (int, float)) or not math.isfinite(value) or value < 0:
                raise ValueError('invalid discard counter')
            k = tuple(sorted(m.get('tags', {}).items()))
            if value < counter.get(k, 0):
                raise ValueError('counter reset')
            counter[k] = value

def allocated():
    global max_allocated
    total = 0
    count = 0
    for d, dirs, files in os.walk(RUN/'data', followlinks=False):
        if len(dirs) > 32:
            raise ValueError('data directory cap')
        for name in files:
            count += 1
            s = (Path(d)/name).lstat()
            if count > 64 or not stat.S_ISREG(s.st_mode):
                raise ValueError('data file cap/type')
            total += s.st_blocks*512
    max_allocated = max(max_allocated, total)
    if total > 268435488+67108864:
        raise ValueError('buffer+overhead disk cap')

def close_stall(now):
    global blocked_since
    if blocked_since is not None:
        a = blocked_since
        b = state['burst_start']
        overlap_start = max(a, b+20)
        overlap_end = min(now, b+80)
        samples = [v for t, v in occupancy if overlap_start <= t <= overlap_end]
        if now-a >= 0.05:
            stalls.append({'start': a-START, 'end': now-START, 'seconds': now-a,
                           'phase': phase(a), 'stall_overlap_seconds': max(0, overlap_end-overlap_start),
                           'peak_occupancy_during_stall': max(samples, default=None)})
        blocked_since = None

def signal_child(sig):
    if time.monotonic() >= END:
        raise TimeoutError('no signal time remaining')
    if p is not None and p.poll() is None:
        if pfd is not None and hasattr(signal, 'pidfd_send_signal'):
            signal.pidfd_send_signal(pfd, sig)
        else:
            p.send_signal(sig)

def wait_child(until):
    while p.poll() is None:
        rem = until-time.monotonic()
        if rem <= 0:
            return False
        io_once(min(0.05, rem))
    return True
